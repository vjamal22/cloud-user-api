resource "aws_iam_role" "lex_service_role" {
  name = "fitness-lex-service-role"

  assume_role_policy = jsonencode({
    Version = "2012-10-17"
    Statement = [
      {
        Effect = "Allow"
        Principal = {
          Service = "lexv2.amazonaws.com"
        }
        Action = "sts:AssumeRole"
      }
    ]
  })
}

resource "aws_lexv2models_bot" "fitness_bot" {
  name                        = "fitness-assistant-bot"
  role_arn                    = aws_iam_role.lex_service_role.arn
  data_privacy {
    child_directed = false
  }

  idle_session_ttl_in_seconds = 300
}

resource "aws_lexv2models_bot_locale" "fitness_bot_locale" {
  bot_id                           = aws_lexv2models_bot.fitness_bot.id
  bot_version                      = "DRAFT"
  locale_id                        = "en_US"

}
